\section{Backtesting (Point 2)}\label{sec:backtesting}

\subsection{Tests and loss functions}
With $n$ backtest days, $n_1$ hits and $\hat\pi = n_1/n$:
\begin{itemize}
  \item \textbf{Unconditional coverage} \citep{kupiec1995}:
        $LR_{UC} = -2\log\frac{(1-\alpha)^{n-n_1}\alpha^{n_1}}{(1-\hat\pi)^{n-n_1}\hat\pi^{n_1}} \sim \chi^2_1$.
  \item \textbf{Independence and conditional coverage} \citep{christoffersen1998}: the hits follow a first-order Markov
        chain with transition probabilities $\pi_{01},\pi_{11}$; $LR_{IND}$ tests $\pi_{01}=\pi_{11}$ ($\chi^2_1$) and
        $LR_{CC} = LR_{UC} + LR_{IND} \sim \chi^2_2$.
  \item \textbf{Berkowitz independence test} \citep{berkowitz2011}: Ljung--Box statistic with 5 lags on the
        demeaned hit sequence $I_t - \alpha$, $\chi^2_5$ under independence.
  \item \textbf{Dynamic Quantile} \citep{engle2004}: regression of $Hit_t = I_t-\alpha$ on a constant, $Hit_{t-1},\dots,Hit_{t-4}$
        and $\VaR_t$; $DQ = \hat{\bm\beta}'\bm X'\bm X\hat{\bm\beta}/(\alpha(1-\alpha)) \sim \chi^2_6$.
  \item \textbf{Diebold--Mariano} \citep{diebold1995}: each model against Historical Simulation, with the tick loss
        $L_t = (\alpha - I_t)(r_t - \VaR_t)$ \citep{gonzalez2004}, strictly consistent for the quantile, and the Lopez
        magnitude loss $L_t = I_t\,[1 + (r_t-\VaR_t)^2]$ \citep{lopez1999}; Newey--West variance.
  \item \textbf{Model Confidence Set} \citep{hansen2011}: $T_{\max}$ statistic on the tick loss, 90\% confidence,
        circular block bootstrap with $B = 1{,}000$ replications and blocks of 10 days.
\end{itemize}
